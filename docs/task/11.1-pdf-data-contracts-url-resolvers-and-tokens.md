### Explicit Non-Goals
- Do **not** build React UI components (`CanvasPdf`, `SourceDocuments`, `PdfPreviewModal`) in this subtask.
- Do **not** introduce `pdfjs-dist` or any external PDF parsing library.
- Do **not** route PDF files through `/api/media/drive/:fileId` (which is strictly an image resizer/proxy); PDFs must resolve to official Google Drive `/preview` and `/view` URLs.
- Do **not** modify any Consumer application files (`package-boundary.md`).

### 3. Google Drive PDF URL Utilities (`src/utils/mediaUrl.ts`)
```ts
export interface ResolvedSinglePdf {
  fileId: string;
  previewUrl: string;
  viewUrl: string;
  rawUrl: string;
}

/**
 * Generates an official Google Drive embedded preview URL for a file ID or Drive file URL.
 * Format: https://drive.google.com/file/d/{fileId}/preview
 */
export function buildDrivePreviewUrl(fileIdOrUrl: string): string {
  const fileId = extractDriveFileId(fileIdOrUrl);
  if (!fileId) return '';
  return `https://drive.google.com/file/d/${fileId}/preview`;
}

/**
 * Generates an official Google Drive view URL for opening a file in a new tab.
 * Format: https://drive.google.com/file/d/{fileId}/view
 */
export function buildDriveViewUrl(fileIdOrUrl: string): string {
  const fileId = extractDriveFileId(fileIdOrUrl);
  if (!fileId) return '';
  return `https://drive.google.com/file/d/${fileId}/view`;
}

/**
 * Resolves any valid Google Drive single-file link variant into canonical
 * iframe preview and new-tab view URLs for PdfElement persistence.
 * Explicitly rejects folder URLs (/folders/).
 */
export function resolveSingleDrivePdfUrl(input: string): ResolvedSinglePdf | null {
  const fileId = extractDriveFileId(input);
  if (!fileId) return null;

  return {
    fileId,
    previewUrl: buildDrivePreviewUrl(fileId),
    viewUrl: buildDriveViewUrl(fileId),
    rawUrl: input.trim(),
  };
}
```

```ts
export const DocumentsElementSchema = z.object({
  id: z.string(),
  type: z.literal('documents'),
  pointId: z.string(),
  title: z.string().optional(),
  titlePointId: z.string().optional(),
  subtitle: z.string().optional(),
  subtitlePointId: z.string().optional(),
  driveUrl: z.string().optional(),
  description: z.string().optional(),
  lastVerified: z.string().optional(),
  syncedDocuments: z.array(z.record(z.string(), z.any())).optional(),
}).passthrough();

export const CanvasLeafElementSchema = z.union([
  ChartElementSchema,
  GalleryElementSchema,
  DocumentsElementSchema,
  z.object({
    id: z.string(),
    type: z.enum(['text', 'photo', 'pdf']),
  }).passthrough(),
]);
```
